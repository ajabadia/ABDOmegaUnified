local omega = require("omega")

-- Define un generador de secuencias
local seq = omega.Seq.new()
    :add{note=36, vel=100, len=0.5}  -- Kick
    :add{note=40, vel=80, len=0.25}  -- Snare
    :repeat(4)
    :mutate(function(evt)
        -- Variación aleatoria con seed determinista
        if math.random() < 0.3 then
            evt.vel = evt.vel * 0.8
        end
        return evt
    end)

-- Modulación procedural: LFO con forma custom
local lfo = omega.LFO.new()
    :waveform(function(phase)
        -- Onda de diente de sierra con glitch aleatorio
        local base = phase % 1.0
        if math.random() < 0.01 then
            return base * 2.0 -- Spike
        end
        return base
    end)
    :rate(omega.BPM.sync(1, 4)) -- 1/4 note

-- Conecta al parámetro
omega.patch.filter.cutoff:modulate(lfo, 0.5)

-- Guarda como preset versionado
omega.preset.save("Algorithmic Dub Bass", {
    algorithmic = true,
    seed = os.time()
})

class VoiceManager
{
    struct VoiceState {
        std::unique_ptr<IVoice> voice;
        std::atomic<uint64_t> timestamp{0};
        std::atomic<bool> active{false};
        int midiNote = -1;
        float velocity = 0.0f;
    };
    
    std::array<VoiceState, 32> voices;
    std::atomic<uint64_t> globalCounter{0};
    std::atomic<int> activeVoiceLimit{16};
    
public:
    void noteOn(int note, float velocity, int channel)
    {
        // 1. Buscar voz inactiva
        for (int i = 0; i < activeVoiceLimit.load(); ++i) {
            if (!voices[i].active.load()) {
                activateVoice(i, note, velocity, channel);
                return;
            }
        }
        
        // 2. Voice stealing: encontrar la más antigua (timestamp más bajo)
        uint64_t oldestTime = UINT64_MAX;
        int oldestIdx = 0;
        
        for (int i = 0; i < activeVoiceLimit.load(); ++i) {
            uint64_t t = voices[i].timestamp.load();
            if (t < oldestTime) {
                oldestTime = t;
                oldestIdx = i;
            }
        }
        
        // 3. Steal con fade out rápido (5-10ms) para evitar click
        voices[oldestIdx].voice->stealNote();
        activateVoice(oldestIdx, note, velocity, channel);
    }
    
private:
    void activateVoice(int idx, int note, float velocity, int channel)
    {
        voices[idx].timestamp.store(globalCounter.fetch_add(1));
        voices[idx].active.store(true);
        voices[idx].midiNote = note;
        voices[idx].velocity = velocity;
        voices[idx].voice->noteOn(note, velocity, channel);
    }
};

enum class ModSource {

enum class ModDestination {
    Off = 0,
    OscPitch, OscLevel, OscShape,
    FilterCutoff, FilterResonance, FilterDrive,
    AmpLevel,
    LFORate, LFODepth,
    FXMix, FXParam1, FXParam2,
    // 64 destinos totales...
    Count = 64
};

struct ModulationRoute {
    ModSource source = ModSource::Off;
    ModDestination dest = ModDestination::Off;
    float amount = 0.0f;        // -1.0 a 1.0
    Curve curve = Curve::Linear;
    bool bipolar = true;
};

class ModulationMatrix
{
    static constexpr int MaxRoutes = 32;
    std::array<ModulationRoute, MaxRoutes> routes;
    
    // Snapshot de valores de fuentes (actualizado una vez por bloque)
    std::array<float, static_cast<int>(ModSource::Count)> sourceValues;
    
public:
    void processBlock(int numSamples)
    {
        // 1. Calcular fuentes (LFOs, envelopes, etc.)
        updateSources(numSamples);
        
        // 2. Aplicar a destinos
        for (const auto& route : routes) {
            if (route.source == ModSource::Off || route.dest == ModDestination::Off)
                continue;
                
            float value = sourceValues[static_cast<int>(route.source)];
            value = applyCurve(value, route.curve);
            value *= route.amount;
            
            applyToDestination(route.dest, value);
        }
    }
};

class EngineRouter : public ISynthesisEngine
{
    std::unique_ptr<ISynthesisEngine> currentEngine;
    std::atomic<bool> pendingEngineChange{false};
    std::unique_ptr<ISynthesisEngine> nextEngine;
    
public:
    void switchEngine(std::unique_ptr<ISynthesisEngine> newEngine)
    {
        // Cambio sin glitch: preparar, luego swap en punto seguro
        newEngine->prepare(sampleRate, samplesPerBlock);
        nextEngine = std::move(newEngine);
        pendingEngineChange.store(true);
    }
    
    void renderNextBlock(AudioBuffer<float>& buffer, MidiBuffer& midi) override
    {
        if (pendingEngineChange.exchange(false)) {
            // Crossfade de 5ms entre engines
            crossfadeEngines(currentEngine.get(), nextEngine.get(), buffer, midi);
            currentEngine = std::move(nextEngine);
        } else {
            currentEngine->renderNextBlock(buffer, midi);
        }
    }
};

class MidiQueue
{
    struct MidiEvent {
        juce::MidiMessage message;
        int samplePosition;
    };
    
    std::array<MidiEvent, 1024> buffer;
    juce::AbstractFifo fifo{1024};
    
public:
    // Llamado desde UI thread o MIDI callback
    void push(const juce::MidiMessage& msg, int samplePos)
    {
        int start1, size1, start2, size2;
        fifo.prepareToWrite(1, start1, size1, start2, size2);
        
        if (size1 > 0) {
            buffer[start1] = {msg, samplePos};
        } else if (size2 > 0) {
            buffer[start2] = {msg, samplePos};
        }
        fifo.finishedWrite(size1 + size2);
    }
    
    // Llamado desde audio thread
    void popAll(juce::MidiBuffer& dest, int numSamples)
    {
        int start1, size1, start2, size2;
        fifo.prepareToRead(1024, start1, size1, start2, size2);
        
        for (int i = 0; i < size1; ++i)
            dest.addEvent(buffer[start1 + i].message, 
                         buffer[start1 + i].samplePosition);
        for (int i = 0; i < size2; ++i)
            dest.addEvent(buffer[start2 + i].message,
                         buffer[start2 + i].samplePosition);
                         
        fifo.finishedRead(size1 + size2);
    }
};
